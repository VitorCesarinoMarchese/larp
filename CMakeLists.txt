cmake_minimum_required(VERSION 3.25)
project(larp VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LARP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
if(LARP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include_directories(${PROJECT_SOURCE_DIR})
enable_testing()
add_executable(protocol-test tests/protocol_test.cpp)
add_test(NAME protocol COMMAND protocol-test)
add_executable(reassembly-test tests/reassembly_test.cpp)
add_test(NAME reassembly COMMAND reassembly-test)
add_executable(udp-test tests/udp_test.cpp)
add_test(NAME udp COMMAND udp-test)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Only the Linux socket backend is implemented")
endif()
add_library(larp-platform platform/udp_linux.cpp)
target_link_libraries(udp-test PRIVATE larp-platform)
add_executable(larp-host host/main.cpp)
add_executable(larp-client client/main.cpp)
target_link_libraries(larp-host PRIVATE larp-platform)
target_link_libraries(larp-client PRIVATE larp-platform)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/integration.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(integration PROPERTIES TIMEOUT 15)
endif()
add_executable(loss-test tests/loss_test.cpp)
add_test(NAME loss COMMAND loss-test)
