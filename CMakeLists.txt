cmake_minimum_required(VERSION 3.25)
project(larp VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LARP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
if(LARP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include_directories(${PROJECT_SOURCE_DIR})
enable_testing()
add_executable(protocol-test tests/protocol_test.cpp)
add_test(NAME protocol COMMAND protocol-test)
add_executable(reassembly-test tests/reassembly_test.cpp)
add_test(NAME reassembly COMMAND reassembly-test)
