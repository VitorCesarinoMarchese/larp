cmake_minimum_required(VERSION 3.25)
project(larp VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LARP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
if(LARP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include_directories(${PROJECT_SOURCE_DIR})
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
enable_testing()
add_executable(h264-test tests/h264_test.cpp)
add_library(larp-codec codec/h264.cpp)
target_link_libraries(larp-codec PRIVATE PkgConfig::FFMPEG)
target_link_libraries(h264-test PRIVATE larp-codec)
add_executable(larp-codec-bench benchmarks/codec.cpp)
target_link_libraries(larp-codec-bench PRIVATE larp-codec)
add_test(NAME h264 COMMAND h264-test)
option(LARP_TEST_NVIDIA "Run tests requiring an NVIDIA GPU and working driver" OFF)
if(LARP_TEST_NVIDIA)
  add_executable(larp-cuda-init benchmarks/cuda_init.cpp)
  target_link_libraries(larp-cuda-init PRIVATE ${CMAKE_DL_LIBS})
  add_executable(larp-nvenc-sessions benchmarks/nvenc_sessions.cpp)
  target_link_libraries(larp-nvenc-sessions PRIVATE PkgConfig::FFMPEG ${CMAKE_DL_LIBS})
  add_test(NAME nvenc COMMAND h264-test nvidia software)
  add_test(NAME nvdec COMMAND h264-test software nvidia)
  add_test(NAME nvidia-roundtrip COMMAND h264-test nvidia nvidia)
  set_tests_properties(nvenc nvdec nvidia-roundtrip PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE LABELS nvidia)
endif()
add_executable(protocol-test tests/protocol_test.cpp)

add_executable(udp-test tests/udp_test.cpp)
add_test(NAME udp COMMAND udp-test)

target_link_libraries(udp-test PRIVATE larp-platform)

add_executable(larp-client client/main.cpp)
target_link_libraries(larp-host PRIVATE larp-session larp-codec)
target_link_libraries(larp-client PRIVATE larp-session larp-codec)
option(LARP_RENDER "Build the SDL3 live preview window" ON)
if(LARP_RENDER)
  pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3>=3.2.0)
  add_library(larp-render render/window_sdl.cpp)
  target_link_libraries(larp-render PRIVATE PkgConfig::SDL3)
  target_link_libraries(larp-client PRIVATE larp-render)
  target_compile_definitions(larp-client PRIVATE LARP_RENDER=1)
  add_executable(render-test tests/render_test.cpp)
  target_link_libraries(render-test PRIVATE larp-render PkgConfig::SDL3)
  add_test(NAME render COMMAND render-test)
  set_tests_properties(render PROPERTIES TIMEOUT 10)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  if(LARP_RENDER)
    add_test(NAME view-client COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/view_client.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(view-client PROPERTIES TIMEOUT 20)
    add_test(NAME view-recovery COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/view_recovery.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(view-recovery PROPERTIES TIMEOUT 20)
  endif()
  add_test(NAME codec-options COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/codec_options.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(codec-options PROPERTIES TIMEOUT 15)
  find_program(FFMPEG_PROGRAM ffmpeg)
  if(FFMPEG_PROGRAM)
    add_test(NAME h264-loss COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_loss.py ${CMAKE_CURRENT_BINARY_DIR} ${FFMPEG_PROGRAM})
  else()
    add_test(NAME h264-loss COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_loss.py ${CMAKE_CURRENT_BINARY_DIR})
  endif()
  set_tests_properties(h264-loss PROPERTIES TIMEOUT 20)
  add_test(NAME h264-client COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_client.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(h264-client PROPERTIES TIMEOUT 15)
  if(LARP_TEST_NVIDIA)
    if(FFMPEG_PROGRAM)
      add_test(NAME nvidia-loss COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_loss.py ${CMAKE_CURRENT_BINARY_DIR} ${FFMPEG_PROGRAM} nvidia nvidia)
    else()
      add_test(NAME nvidia-loss COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_loss.py ${CMAKE_CURRENT_BINARY_DIR} - nvidia nvidia)
    endif()
    set_tests_properties(nvidia-loss PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE LABELS nvidia)
    add_test(NAME nvidia-client COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_client.py ${CMAKE_CURRENT_BINARY_DIR} nvidia nvidia)
    set_tests_properties(nvidia-client PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE LABELS nvidia)
  endif()
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/integration.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(integration PROPERTIES TIMEOUT 15)
endif()
