cmake_minimum_required(VERSION 3.25)
project(larp VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LARP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
if(LARP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include_directories(${PROJECT_SOURCE_DIR})
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
enable_testing()
add_executable(h264-test tests/h264_test.cpp)
add_library(larp-codec codec/h264.cpp)
target_link_libraries(larp-codec PRIVATE PkgConfig::FFMPEG)
target_link_libraries(h264-test PRIVATE larp-codec)
add_test(NAME h264 COMMAND h264-test)
add_executable(protocol-test tests/protocol_test.cpp)
add_test(NAME protocol COMMAND protocol-test)
add_executable(reassembly-test tests/reassembly_test.cpp)
add_test(NAME reassembly COMMAND reassembly-test)
add_executable(udp-test tests/udp_test.cpp)
add_test(NAME udp COMMAND udp-test)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Only the Linux socket backend is implemented")
endif()
add_library(larp-platform platform/udp_linux.cpp)
target_link_libraries(udp-test PRIVATE larp-platform)
add_executable(larp-host host/main.cpp host/h264_main.cpp)
add_executable(larp-client client/main.cpp)
target_link_libraries(larp-host PRIVATE larp-platform larp-codec)
target_link_libraries(larp-client PRIVATE larp-platform larp-codec)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  find_program(FFMPEG_PROGRAM ffmpeg)
  if(FFMPEG_PROGRAM)
    add_test(NAME h264-loss COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_loss.py ${CMAKE_CURRENT_BINARY_DIR} ${FFMPEG_PROGRAM})
  else()
    add_test(NAME h264-loss COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_loss.py ${CMAKE_CURRENT_BINARY_DIR})
  endif()
  set_tests_properties(h264-loss PROPERTIES TIMEOUT 20)
  add_test(NAME h264-client COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/h264_client.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(h264-client PROPERTIES TIMEOUT 15)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/integration.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(integration PROPERTIES TIMEOUT 15)
endif()
add_executable(loss-test tests/loss_test.cpp)
add_test(NAME loss COMMAND loss-test)
add_executable(pixels-test tests/pixels_test.cpp)
add_test(NAME pixels COMMAND pixels-test)
option(LARP_CAPTURE "Build Linux PipeWire capture" ON)
if(LARP_CAPTURE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
  add_library(larp-capture capture/linux/portal.cpp capture/linux/pipewire_capture.cpp)
  target_link_libraries(larp-capture PRIVATE PkgConfig::PIPEWIRE PkgConfig::GIO)
  add_executable(larp-capture-probe capture/probe.cpp)
  target_link_libraries(larp-capture-probe PRIVATE larp-capture)
endif()
add_executable(raw-frame-test tests/raw_frame_test.cpp)
add_test(NAME raw-frame COMMAND raw-frame-test)
if(Python3_FOUND)
  add_test(NAME raw-client COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/raw_client.py ${CMAKE_CURRENT_BINARY_DIR})
  set_tests_properties(raw-client PROPERTIES TIMEOUT 10)
endif()
if(LARP_CAPTURE)
  target_sources(larp-host PRIVATE host/capture_main.cpp)
  target_compile_definitions(larp-host PRIVATE LARP_CAPTURE=1)
  target_link_libraries(larp-host PRIVATE larp-capture)
  if(Python3_FOUND)
    add_test(NAME capture-startup COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/capture_startup.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(capture-startup PROPERTIES TIMEOUT 10)
  endif()
endif()
if(LARP_CAPTURE)
  add_executable(pipewire-context-test tests/pipewire_context_test.cpp)
  target_link_libraries(pipewire-context-test PRIVATE PkgConfig::PIPEWIRE)
  add_test(NAME pipewire-context COMMAND pipewire-context-test)
endif()
